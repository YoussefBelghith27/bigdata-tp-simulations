import csv
#--
brut = sc.textFile("/user/student/input/restaurants.csv")
#--
entete = brut.first()
#--
donnees = brut.filter(lambda l: l != entete)
#--
def parse(ligne):
    return next(csv.reader([ligne]))
#--
restos = donnees.map(parse)
#--
print(restos.take(2))
#--
par_ville = restos.map(lambda r: (r[2], 1)).reduceByKey(lambda a, b: a + b)
#--
print(par_ville.sortBy(lambda kv: -kv[1]).collect())
#--
par_cuisine = restos.map(lambda r: (r[3], 1)).reduceByKey(lambda a, b: a + b)
#--
print(par_cuisine.sortBy(lambda kv: -kv[1]).collect())
#--
def valide(r):
    try:
        float(r[4]); return True
    except (ValueError, IndexError):
        return False
#--
valides = restos.filter(valide)
#--
somme = valides.map(lambda r: (r[3], (float(r[4]), 1))) \
               .reduceByKey(lambda a, b: (a[0] + b[0], a[1] + b[1]))
#--
moyennes = somme.mapValues(lambda s: round(s[0] / s[1], 2))
#--
print(moyennes.sortBy(lambda kv: -kv[1]).collect())
#--

import time
#--
from pyspark import StorageLevel
#--
def chrono(rdd):
    t0 = time.time()
    n = rdd.count()
    return n, round(time.time() - t0, 3)
#--
print("sans cache :", chrono(valides), chrono(valides))
#--
valides.cache()
#--
print("cache()    :", chrono(valides), chrono(valides))
